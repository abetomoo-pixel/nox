## e. payroll_reopen（pg_get_functiondef 逐語）
  1: CREATE OR REPLACE FUNCTION public.payroll_reopen(p_org_id uuid, p_actor uuid, p_run_id uuid, p_idem_key uuid, p_reason text)
  2:  RETURNS text
  3:  LANGUAGE plpgsql
  4:  SECURITY DEFINER
  5:  SET search_path TO 'public'
  6: AS $function$
  7: declare
  8:   v_org         uuid;
  9:   v_store       uuid;
 10:   v_status      text;
 11:   v_reopen_idem uuid;
 12:   v_fin_idem    uuid;
 13:   v_old_ps      date;
 14:   v_old_pe      date;
 15:   v_retired     jsonb;
 16:   v_arrec   jsonb;
 17:   v_advrec  jsonb;
 18:   v_okrec   jsonb;
 19:   v_rolled      jsonb;
 20:   v_rolled_adv  jsonb;
 21:   v_rolled_ok   jsonb;
 22:   v_flag        boolean;
 23: begin
 24:   select org_id, store_id, status, reopen_idem_key, finalize_idem_key, period_start, period_end
 25:     into v_org, v_store, v_status, v_reopen_idem, v_fin_idem, v_old_ps, v_old_pe
 26:     from public.payroll_runs where id = p_run_id;
 27:   if v_org is null then raise exception 'run not found'; end if;
 28:   if p_org_id is null or v_org <> p_org_id then raise exception 'forbidden'; end if;
 29: 
 30:   -- reopen_flow(店舗行 → org 行 → false)
 31:   select f.enabled into v_flag from public.feature_flags f where f.org_id = v_org and f.store_id = v_store and f.key = 'reopen_flow';
 32:   if v_flag is null then
 33:     select f.enabled into v_flag from public.feature_flags f where f.org_id = v_org and f.store_id is null and f.key = 'reopen_flow';
 34:   end if;
 35:   if not coalesce(v_flag, false) then raise exception 'feature_disabled:reopen_flow'; end if;
 36:   if p_reason is null or length(trim(p_reason)) not between 1 and 200 then raise exception 'reason_required'; end if;
 37: 
 38:   if v_status = 'paid' then raise exception 'run paid'; end if;
 39: 
 40:   if p_idem_key is not null and v_status = 'draft' and v_reopen_idem is not distinct from p_idem_key then
 41:     return 'draft';
 42:   end if;
 43: 
 44:   if v_status <> 'finalized' then raise exception 'not finalized'; end if;
 45: 
 46:   if exists (select 1 from public.payment_records pr where pr.run_id = p_run_id) then
 47:     raise exception 'payments exist';
 48:   end if;
 49: 
 50:   select jsonb_agg(jsonb_build_object('cast_id', ps.cast_id, 'net', ps.net, 'breakdown', ps.breakdown_json))
 51:     into v_retired from public.payslips ps where ps.run_id = p_run_id;
 52: 
 53:   v_rolled := '[]'::jsonb;
 54:   for v_arrec in
 55:     select ae from public.payslips ps,
 56:       lateral jsonb_array_elements(coalesce(ps.breakdown_json->'ar', '[]'::jsonb)) ae
 57:     where ps.run_id = p_run_id
 58:   loop
 59:     update public.receivables r
 60:        set status = v_arrec->>'prev_status',
 61:            deduct_period = nullif(v_arrec->>'prev_deduct_period', ''),
 62:            deducted_amount = (v_arrec->>'prev_deducted_amount')::int
 63:      where r.id = (v_arrec->>'receivable_id')::uuid
 64:        and r.status = v_arrec->>'applied_status'
 65:        and r.deducted_amount = (v_arrec->>'applied_deducted_amount')::int
 66:        and r.deduct_period is not distinct from nullif(v_arrec->>'applied_deduct_period', '');
 67:     if found then v_rolled := v_rolled || v_arrec; end if;
 68:   end loop;
 69:   v_rolled_adv := '[]'::jsonb;
 70:   for v_advrec in
 71:     select ae from public.payslips ps,
 72:       lateral jsonb_array_elements(coalesce(ps.breakdown_json->'adv', '[]'::jsonb)) ae
 73:     where ps.run_id = p_run_id
 74:   loop
 75:     update public.advances a
 76:        set status = v_advrec->>'prev_status',
 77:            deduct_period = nullif(v_advrec->>'prev_deduct_period', ''),
 78:            deducted_amount = (v_advrec->>'prev_deducted_amount')::int
 79:      where a.id = (v_advrec->>'advance_id')::uuid
 80:        and a.status = v_advrec->>'applied_status'
 81:        and a.deducted_amount = (v_advrec->>'applied_deducted_amount')::int
 82:        and a.deduct_period is not distinct from nullif(v_advrec->>'applied_deduct_period', '');
 83:     if found then v_rolled_adv := v_rolled_adv || v_advrec; end if;
 84:   end loop;
 85:   v_rolled_ok := '[]'::jsonb;
 86:   for v_okrec in
 87:     select ae from public.payslips ps,
 88:       lateral jsonb_array_elements(coalesce(ps.breakdown_json->'okuri', '[]'::jsonb)) ae
 89:     where ps.run_id = p_run_id
 90:   loop
 91:     update public.transport t
 92:        set status = v_okrec->>'prev_status',
 93:            deducted_amount = (v_okrec->>'prev_deducted_amount')::int
 94:      where t.id = (v_okrec->>'transport_id')::uuid
 95:        and t.status = v_okrec->>'applied_status'
 96:        and t.deducted_amount = (v_okrec->>'applied_deducted_amount')::int;
 97:     if found then v_rolled_ok := v_rolled_ok || v_okrec; end if;
 98:   end loop;
 99: 
100:   delete from public.payslips where run_id = p_run_id;
101: 
102:   update public.payroll_runs
103:      set status = 'draft', finalized_at = null, finalize_idem_key = null,
104:          period_start = null, period_end = null,
105:          reopen_idem_key = p_idem_key
106:    where id = p_run_id;
107: 
108:   perform public.audit_log_write_service(v_org, p_actor, 'payroll_reopen',
109:     'payroll_runs:' || p_run_id::text,
110:     jsonb_build_object('retired_payslips', coalesce(v_retired, '[]'::jsonb),
111:                        'old_finalize_idem_key', v_fin_idem,
112:                        'old_period_start', v_old_ps, 'old_period_end', v_old_pe,
113:                        'rolled_back_receivables', v_rolled,
114:                        'rolled_back_advances', v_rolled_adv,
115:                        'rolled_back_transport', v_rolled_ok),
116:     jsonb_build_object('status', 'draft', 'reopen_idem_key', p_idem_key),
117:     v_store, p_reason);
118:   return 'reopened';
119: end $function$

## set_store_cast_register 全文（1 本）
  1: CREATE OR REPLACE FUNCTION public.set_store_cast_register(p_store_id uuid, p_enabled boolean)
  2:  RETURNS void
  3:  LANGUAGE plpgsql
  4:  SECURITY DEFINER
  5:  SET search_path TO 'public'
  6: AS $function$
  7: declare
  8:   v_store record;
  9:   v_prev  boolean;
 10: begin
 11:   if public.auth_org_id() is null then raise exception 'forbidden'; end if;
 12:   if not public.billing_writable_of(public.auth_org_id()) then raise exception 'billing locked'; end if;
 13:   if p_enabled is null then raise exception 'bad enabled'; end if;
 14:   select id, org_id, settings_json into v_store from public.stores where id = p_store_id;
 15:   if v_store.org_id is null or v_store.org_id <> public.auth_org_id() then raise exception 'forbidden'; end if;
 16:   if public.auth_role() <> 'owner' then raise exception 'forbidden'; end if;  -- 店ポリシー＝owner 限定（okuri_mode と同格）
 17: 
 18:   v_prev := coalesce(v_store.settings_json->>'cast_register_enabled', '') = 'true';
 19:   update public.stores
 20:      set settings_json = jsonb_set(coalesce(settings_json, '{}'::jsonb), '{cast_register_enabled}', to_jsonb(p_enabled), true)
 21:    where id = p_store_id;
 22: 
 23:   perform public.audit_log_write('set_store_cast_register', 'stores:' || p_store_id::text,
 24:     jsonb_build_object('cast_register_enabled', v_prev), jsonb_build_object('cast_register_enabled', p_enabled), p_store_id);
 25: end $function$

## set_store_receipt_profile 全文（1 本）
  1: CREATE OR REPLACE FUNCTION public.set_store_receipt_profile(p_store_id uuid, p_address text, p_tel text, p_reg_no text, p_footer text)
  2:  RETURNS void
  3:  LANGUAGE plpgsql
  4:  SECURITY DEFINER
  5:  SET search_path TO 'public'
  6: AS $function$
  7: declare
  8:   v_org     uuid := public.auth_org_id();
  9:   v_store   record;
 10:   v_addr    text := trim(coalesce(p_address, ''));
 11:   v_tel     text := trim(coalesce(p_tel, ''));
 12:   v_reg     text := trim(coalesce(p_reg_no, ''));
 13:   v_footer  text := trim(coalesce(p_footer, ''));
 14:   v_before  jsonb;
 15: begin
 16:   if v_org is null then raise exception 'forbidden'; end if;
 17:   if not public.billing_writable_of(v_org) then raise exception 'billing locked'; end if;
 18:   if public.auth_role() <> 'owner' then raise exception 'forbidden'; end if;
 19:   if length(v_addr) > 200 then raise exception 'bad address'; end if;
 20:   if length(v_tel) > 50 then raise exception 'bad tel'; end if;
 21:   if length(v_footer) > 200 then raise exception 'bad footer'; end if;
 22:   if v_reg <> '' and v_reg !~ '^T[0-9]{13}$' then raise exception 'bad reg_no'; end if;
 23:   select id, org_id, settings_json into v_store from public.stores where id = p_store_id;
 24:   if v_store.org_id is null or v_store.org_id <> v_org then raise exception 'forbidden'; end if;
 25: 
 26:   v_before := jsonb_build_object(
 27:     'receipt_address', coalesce(v_store.settings_json->>'receipt_address', ''),
 28:     'receipt_tel',     coalesce(v_store.settings_json->>'receipt_tel', ''),
 29:     'invoice_reg_no',  coalesce(v_store.settings_json->>'invoice_reg_no', ''),
 30:     'receipt_footer',  coalesce(v_store.settings_json->>'receipt_footer', '')
 31:   );
 32:   update public.stores
 33:      set settings_json =
 34:        jsonb_set(
 35:          jsonb_set(
 36:            jsonb_set(
 37:              jsonb_set(coalesce(settings_json, '{}'::jsonb),
 38:                '{receipt_address}', to_jsonb(v_addr), true),
 39:              '{receipt_tel}',     to_jsonb(v_tel),    true),
 40:            '{invoice_reg_no}',  to_jsonb(v_reg),    true),
 41:          '{receipt_footer}',  to_jsonb(v_footer), true)
 42:    where id = p_store_id;
 43:   perform public.audit_log_write('set_store_receipt_profile', 'stores:' || p_store_id::text,
 44:     v_before,
 45:     jsonb_build_object('receipt_address', v_addr, 'receipt_tel', v_tel,
 46:                        'invoice_reg_no', v_reg, 'receipt_footer', v_footer),
 47:     p_store_id);
 48: end $function$

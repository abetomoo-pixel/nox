   173	-- ⑥ receivable_collect（回収消込・owner/manager・全額回収・冪等・audit・設計1 案1-A）

   177	create or replace function public.receivable_collect(
   178	  p_receivable_id uuid,
   179	  p_biz_date      date,
   180	  p_method        text default 'cash',
   181	  p_note          text default null,
   182	  p_idem_key      uuid default null
   183	) returns uuid language plpgsql security definer set search_path = public as $$
   184	declare
   185	  v_recv   record;
   186	  v_method text;
   187	  v_actor  uuid;
   188	  v_id     uuid;
   189	begin
   190	  if public.auth_org_id() is null then raise exception 'forbidden'; end if;
   191	  if p_biz_date is null then raise exception 'bad date'; end if;
   192	  if p_idem_key is null then raise exception 'idem required'; end if;
   193	  v_method := coalesce(nullif(trim(coalesce(p_method,'')),''), 'cash');
   194	  if v_method not in ('cash','card','other') then raise exception 'bad method'; end if;
   195	
   196	  -- gate（org 照合 → owner/manager 自店）
   197	  select * into v_recv from public.receivables where id = p_receivable_id;
   198	  if v_recv.id is null or v_recv.org_id <> public.auth_org_id() then raise exception 'forbidden'; end if;
   199	  if not (public.auth_role() = 'owner'
   200	          or (public.auth_role() = 'manager' and v_recv.store_id = public.auth_store_id())) then
   201	    raise exception 'forbidden';
   202	  end if;
   203	
   204	  -- ★冪等・並行リプレイ解決は payment_record_add の live prosrc と同型（前段 idem チェック →
   205	  --   FOR UPDATE 直列化点 → INSERT on conflict do nothing → fallback SELECT）。相違は直列化点のみ＝
   206	  --   payment_record_add は payslip(run,cast) を、本 RPC は receivable 行をロックする。
   207	  -- 冪等（前段・org/ロール照合の後＝org 外ユーザーのキー存在確認悪用を防ぐ）: 既存回収は返す

   212	  select * into v_recv from public.receivables where id = p_receivable_id for update;

   214	  --   コミット済回収がここで必ず可視になる（READ COMMITTED での戻り値欠落＝並行リプレイの穴を封鎖）。
   215	  select id into v_id from public.ar_collections where idem_key = p_idem_key;
   216	  if v_id is not null then return v_id; end if;
   217	
   218	  -- 消込は open のみ（回収済/天引き済/void は不可＝別 idem の二重回収を拒否）
   219	  if v_recv.status <> 'open' then raise exception 'not open'; end if;
   220	
   221	  select id into v_actor from public.users where auth_user_id = auth.uid() and is_active;
   222	  insert into public.ar_collections
   223	    (org_id, store_id, receivable_id, cast_id, customer_id, biz_date, amount, method, note, idem_key, created_by)
   224	  values
   225	    (v_recv.org_id, v_recv.store_id, p_receivable_id, v_recv.cast_id, v_recv.customer_id,
   226	     p_biz_date, v_recv.amount, v_method, nullif(trim(coalesce(p_note,'')),''), p_idem_key, v_actor)
   227	  on conflict (idem_key) do nothing
   228	  returning id into v_id;

   230	    -- ロック内再チェック済ゆえ通常不到達・belt-and-suspenders（payment_record_add 同型）
   231	    select id into v_id from public.ar_collections where idem_key = p_idem_key; return v_id;
   232	  end if;
   233	
   234	  update public.receivables set status = 'collected' where id = p_receivable_id and status = 'open';
   235	
   236	  perform public.audit_log_write('receivable_collect', 'receivables:' || p_receivable_id::text,
   237	    jsonb_build_object('status', 'open', 'deducted_amount', v_recv.deducted_amount),
   238	    jsonb_build_object('status', 'collected', 'collection_id', v_id, 'biz_date', p_biz_date,
   239	                       'amount', v_recv.amount, 'method', v_method),
   240	    v_recv.store_id);
   241	  return v_id;
   242	end $$;

   252	  p_receivable_id uuid,
   253	  p_consent       boolean,
   254	  p_note          text default null
   255	) returns uuid language plpgsql security definer set search_path = public as $$
   256	declare
   257	  v_recv  record;
   258	  v_actor uuid;

   260	  if public.auth_org_id() is null then raise exception 'forbidden'; end if;

   291	-- ⑧ daily_report_aggregate 改修（live 写経＋ar_collected 1源追加・biz_date 直・method='cash' のみ）
   292	--    STABLE・内部専用（4ロール revoke）不変。checks/payments 集計は逐語不変（money-core 非接触）。
   293	-- ══════════════════════════════════════════════════════════════
   294	create or replace function public.daily_report_aggregate(
   295	  p_store_id  uuid,
   296	  p_biz_date  date,
   297	  p_cutoff_hm text,
   298	  p_tax_rate  int
   299	) returns jsonb language plpgsql stable security definer set search_path = public as $$
   300	declare
   301	  v_org   uuid;
   302	  v_start timestamptz;
   303	  v_end   timestamptz;
   304	  v jsonb;
   305	begin
   306	  select org_id into v_org from public.stores where id = p_store_id;
   307	  if v_org is null then raise exception 'not found'; end if;
   308	  -- [D cutoff JST, D+1 cutoff JST)
   309	  v_start := ((p_biz_date::text || ' ' || p_cutoff_hm) )::timestamp at time zone 'Asia/Tokyo';
   310	  v_end   := (((p_biz_date + 1)::text || ' ' || p_cutoff_hm))::timestamp at time zone 'Asia/Tokyo';
   311	  select jsonb_build_object(
   312	    'open_checks', (select count(*) from public.checks c
   313	                     where c.org_id = v_org and c.store_id = p_store_id and c.status = 'open'
   314	                       and c.started_at >= v_start and c.started_at < v_end),
   315	    'slips',  (select count(*) from public.checks c
   316	                where c.org_id = v_org and c.store_id = p_store_id and c.status = 'closed'
   317	                  and c.started_at >= v_start and c.started_at < v_end),
   318	    'guests', (select coalesce(sum(c.people), 0) from public.checks c
   319	                where c.org_id = v_org and c.store_id = p_store_id and c.status = 'closed'
   320	                  and c.started_at >= v_start and c.started_at < v_end),
   321	    'dohan_checks', (select count(*) from public.checks c
   322	                where c.org_id = v_org and c.store_id = p_store_id and c.status = 'closed' and c.nom_type = 'dohan'
   323	                  and c.started_at >= v_start and c.started_at < v_end),
   324	    'cash',  (select coalesce(sum(p.amount), 0) from public.payments p
   325	               join public.checks c on c.id = p.check_id
   326	               where c.org_id = v_org and p.org_id = v_org
   327	                 and c.store_id = p_store_id and c.status = 'closed' and p.method = 'cash'
   328	                 and c.started_at >= v_start and c.started_at < v_end),
   329	    'card',  (select coalesce(sum(p.amount), 0) from public.payments p
   330	               join public.checks c on c.id = p.check_id
   331	               where c.org_id = v_org and p.org_id = v_org
   332	                 and c.store_id = p_store_id and c.status = 'closed' and p.method = 'card'
   333	                 and c.started_at >= v_start and c.started_at < v_end),
   334	    'uri',   (select coalesce(sum(p.amount), 0) from public.payments p
   335	               join public.checks c on c.id = p.check_id
   336	               where c.org_id = v_org and p.org_id = v_org
   337	                 and c.store_id = p_store_id and c.status = 'closed' and p.method = 'ar'
   338	                 and c.started_at >= v_start and c.started_at < v_end),
   339	    'other', (select coalesce(sum(p.amount), 0) from public.payments p
   340	               join public.checks c on c.id = p.check_id
   341	               where c.org_id = v_org and p.org_id = v_org
   342	                 and c.store_id = p_store_id and c.status = 'closed' and p.method = 'other'
   343	                 and c.started_at >= v_start and c.started_at < v_end),
   344	    'drink_sales', (select coalesce(sum(l.line_total), 0) from public.check_lines l
   345	               join public.checks c on c.id = l.check_id
   346	               where c.org_id = v_org and l.org_id = v_org
   347	                 and c.store_id = p_store_id and c.status = 'closed' and l.kind in ('drink','champ')
   348	                 and c.started_at >= v_start and c.started_at < v_end),
   349	    -- ★B6（mig0055）: 回収現金（別掲・biz_date 直・method='cash' のみ＝理論在高加算対象）。
   350	    --   checks/payments 非依存＝発生日 uri との二重計上は起きない（別経路・突合は receivables 直 SELECT）。
   351	    'ar_collected', (select coalesce(sum(x.amount), 0) from public.ar_collections x
   352	               where x.org_id = v_org and x.store_id = p_store_id
   353	                 and x.biz_date = p_biz_date and x.method = 'cash')
   354	  ) into v;
   355	  return v || jsonb_build_object('card_tax', round(((v->>'card')::int) * p_tax_rate / 100.0)::int);
   356	end $$;
   357	revoke execute on function public.daily_report_aggregate(uuid, date, text, int)
   358	  from public, anon, authenticated, service_role;

   360	-- ══════════════════════════════════════════════════════════════
   361	-- ⑨ daily_report_close 改修（live 写経＋ar_collected 凍結＋diff 式へ加算）
   362	--    ★diff = counted − (float + cash + ar_collected − expense − payout)（理論在高に回収現金を含める）。
   363	--    ar_collected=0 の日は従前の diff と一致（後方互換）。税率は stores.card_tax_rate 列読み（mig0051・live）。
   364	-- ══════════════════════════════════════════════════════════════
   365	create or replace function public.daily_report_close(
   366	  p_store_id     uuid,
   367	  p_biz_date     date,
   368	  p_expense      int default 0,
   369	  p_cash_payout  int default 0,
   370	  p_cash_float   int default 0,
   371	  p_counted_cash int default null,
   372	  p_note         text default null,
   373	  p_force        boolean default false,
   374	  p_idem_key     uuid default null
   375	) returns uuid language plpgsql security definer set search_path = public as $$
   376	declare
   377	  v_owner uuid; v_settings jsonb; v_cutoff text; v_rate int;
   378	  v_exist record; v_agg jsonb; v_actor uuid; v_id uuid; v_diff int; v_ar int;
   379	begin
   380	  if public.auth_org_id() is null then raise exception 'forbidden'; end if;
   381	  if p_biz_date is null then raise exception 'bad date'; end if;
   382	  if coalesce(p_expense, -1) < 0 or coalesce(p_cash_payout, -1) < 0 or coalesce(p_cash_float, -1) < 0 then
   383	    raise exception 'bad amount';
   384	  end if;
   385	  if p_counted_cash is not null and p_counted_cash < 0 then raise exception 'bad amount'; end if;
   386	  -- E1 mig0051: 税率は stores.card_tax_rate 列読み（列 CHECK 0..100 が構造保証・既定 5 は列 default と同値）
   387	  select org_id, settings_json, card_tax_rate into v_owner, v_settings, v_rate
   388	    from public.stores where id = p_store_id;
   389	  if v_owner is null or v_owner <> public.auth_org_id() then raise exception 'forbidden'; end if;
   390	  if not (public.auth_role() = 'owner'
   391	          or (public.auth_role() = 'manager' and p_store_id = public.auth_store_id())) then
   392	    raise exception 'forbidden';
   393	  end if;
   394	
   395	  -- 冪等: 同一 (store, biz_date) の既存行＝同一キーなら成功・別キーは reclose を促す
   396	  select * into v_exist from public.daily_reports
   397	    where store_id = p_store_id and biz_date = p_biz_date;
   398	  if v_exist.id is not null then
   399	    if p_idem_key is not null and v_exist.close_idem_key = p_idem_key then return v_exist.id; end if;
   400	    raise exception 'already closed';
   401	  end if;
   402	
   403	  -- 設定スナップショット（cutoff 既定 06:00＝json のまま／税率＝列読み・raise は防御深度で残置）
   404	  v_cutoff := coalesce(nullif(trim(coalesce(v_settings, '{}'::jsonb)->>'biz_cutoff_hm'), ''), '06:00');
   405	  if v_cutoff !~ '^([01][0-9]|2[0-3]):[0-5][0-9]$' or v_rate < 0 then
   406	    raise exception 'bad store settings';
   407	  end if;
   408	
   409	  v_agg := public.daily_report_aggregate(p_store_id, p_biz_date, v_cutoff, v_rate);
   410	  v_ar  := (v_agg->>'ar_collected')::int;
   411	
   412	  -- 【決定1】open 伝票が範囲内に残る場合は既定拒否・p_force で強行（残数を記録）
   413	  if (v_agg->>'open_checks')::int > 0 and not p_force then
   414	    raise exception 'open checks remain';
   415	  end if;
   416	
   417	  -- 【決定2＋B6】diff = counted − (float + cash + ar_collected − expense − payout)
   418	  --   （モック H=Oi−q に回収現金を理論在高へ加算＝ドロワー実査整合）。counted 未入力時は null。
   419	  v_diff := case when p_counted_cash is null then null
   420	                 else p_counted_cash - (coalesce(p_cash_float,0) + (v_agg->>'cash')::int + v_ar
   421	                                        - coalesce(p_expense,0) - coalesce(p_cash_payout,0)) end;
   422	
   423	  select id into v_actor from public.users where auth_user_id = auth.uid() and is_active;
   424	  insert into public.daily_reports
   425	    (org_id, store_id, biz_date,
   426	     cash, card_gross, card_tax, uri, other, drink_sales, dohan_checks, slips, guests,
   427	     open_checks_count, ar_collected, expense, cash_payout, cash_float, counted_cash, diff, note,
   428	     biz_cutoff_hm, card_tax_rate, close_idem_key, closed_by)
   429	  values
   430	    (public.auth_org_id(), p_store_id, p_biz_date,
   431	     (v_agg->>'cash')::int, (v_agg->>'card')::int, (v_agg->>'card_tax')::int,
   432	     (v_agg->>'uri')::int, (v_agg->>'other')::int, (v_agg->>'drink_sales')::int,
   433	     (v_agg->>'dohan_checks')::int, (v_agg->>'slips')::int, (v_agg->>'guests')::int,
   434	     (v_agg->>'open_checks')::int, v_ar,
   435	     coalesce(p_expense,0), coalesce(p_cash_payout,0), coalesce(p_cash_float,0),
   436	     p_counted_cash, v_diff, p_note,
   437	     v_cutoff, v_rate, p_idem_key, v_actor)
   438	  returning id into v_id;
   439	  perform public.audit_log_write('daily_report_close', 'daily_reports:' || v_id::text, null,
   440	    (select to_jsonb(d) from public.daily_reports d where d.id = v_id), p_store_id);
   441	  return v_id;
   442	end $$;

   446	-- ══════════════════════════════════════════════════════════════
   447	-- ⑩ daily_report_reclose 改修（live 写経＋ar_collected 再集計＋diff 式へ加算）
   448	--    ★締め後に遅れて回収した分は reclose で ar_collected に拾われる（設計2 の遅延回収経路）。
   449	-- ══════════════════════════════════════════════════════════════
   450	create or replace function public.daily_report_reclose(
   451	  p_report_id    uuid,
   452	  p_expense      int default null,
   453	  p_cash_payout  int default null,
   454	  p_cash_float   int default null,
   455	  p_counted_cash int default null,
   456	  p_note         text default null,
   457	  p_force        boolean default false
   458	) returns uuid language plpgsql security definer set search_path = public as $$
   459	declare
   460	  v_row record; v_agg jsonb; v_before jsonb; v_diff int; v_ar int;
   461	  v_expense int; v_payout int; v_float int; v_counted int;
   462	begin
   463	  if public.auth_org_id() is null then raise exception 'forbidden'; end if;
   464	  select * into v_row from public.daily_reports where id = p_report_id;
   465	  if v_row.id is null or v_row.org_id <> public.auth_org_id() then raise exception 'forbidden'; end if;
   466	  if not (public.auth_role() = 'owner'
   467	          or (public.auth_role() = 'manager' and v_row.store_id = public.auth_store_id())) then
   468	    raise exception 'forbidden';
   469	  end if;
   470	  v_before := to_jsonb(v_row);
   471	
   472	  -- 再集計は凍結済みの cutoff / rate を使う（範囲定義・税率は初回締めから不変）
   473	  v_agg := public.daily_report_aggregate(v_row.store_id, v_row.biz_date, v_row.biz_cutoff_hm, v_row.card_tax_rate);
   474	  v_ar  := (v_agg->>'ar_collected')::int;
   475	  if (v_agg->>'open_checks')::int > 0 and not p_force then
   476	    raise exception 'open checks remain';
   477	  end if;
   478	
   479	  v_expense := coalesce(p_expense, v_row.expense);
   480	  v_payout  := coalesce(p_cash_payout, v_row.cash_payout);
   481	  v_float   := coalesce(p_cash_float, v_row.cash_float);
   482	  v_counted := coalesce(p_counted_cash, v_row.counted_cash);
   483	  if v_expense < 0 or v_payout < 0 or v_float < 0 or (v_counted is not null and v_counted < 0) then
   484	    raise exception 'bad amount';
